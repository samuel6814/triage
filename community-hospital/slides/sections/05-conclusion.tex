\section{Conclusion}
% =============================================================================

\begin{frame}{Summary}
  \small
  \begin{enumerate}
    \item \textbf{Research gaps:} estimate $\hat{c}$ from $X$ when $\mathbf{v}=\emptyset$; bind $C$ to $P(C)$; model ED flow as transshipment LP \citep{Stewart2023, Sundstrom2014, Mitchell2023}.
    \item \textbf{Community Hospital:} 1000 skewed cases; 97.5\% colour accuracy; pathway cards on every row \citep{thisstudy2026, SATS2012}.
    \item \textbf{Transshipment OR:} 62\% cost reduction, 44\% acute-load reduction with chatbot Green diversion \citep{thisstudy2026, Yilmaz2021}.
    \item \textbf{NLP design:} conversational intake, constrained output (not free-form diagnosis) \citep{Stewart2023, Lee2020, Cheng2025}.
  \end{enumerate}
\end{frame}

\begin{frame}{Limitations}
  \small
  \begin{itemize}
    \item Synthetic cohort: not yet prospective KATH validation \citep{thisstudy2026, Rominski2014}.
    \item Simulated chatbot eval when ML service offline (8k holdout 99.92\% reference) \citep{thisstudy2026, Lee2020}.
    \item Transshipment is an analytical model, not a deployed optimizer \citep{thisstudy2026, Mitchell2023}.
  \end{itemize}
\end{frame}
