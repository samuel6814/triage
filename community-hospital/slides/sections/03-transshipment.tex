\section{Transshipment optimization}
% =============================================================================

\begin{frame}{ED overcrowding as a network problem}
  \small
  \textbf{Problem:} long waits and acute-zone congestion when non-urgent patients enter the wrong queue \citep{Yilmaz2021, Mitchell2023}.

  \vspace{0.15cm}
  \textbf{OR tool:} transshipment LP
  \[
    \min Z = \sum_{(i,j)} c_{ij}\, x_{ij}
  \]
  subject to supply, balance, and demand constraints.

  \plain{Patients are flow; arcs are routing paths; cost $c_{ij}$ is expected wait plus wrong-queue penalty \citep{thisstudy2026}.}
\end{frame}

\begin{frame}{LP structure}
  \footnotesize
  \textbf{Decision variables:} $x_{ij}$ = patients routed on arc $(i,j)$.

  \vspace{0.08cm}
  \textbf{Supply:} arrivals (50, 150, 300, 500) \citep{thisstudy2026}.

  \textbf{Balance:} inflow $=$ outflow at chatbot and desk.

  \textbf{Demand:} resuscitation 50, majors 450, OPD 500 \citep{thisstudy2026}.

  \vspace{0.1cm}
  \begin{block}{Community Hospital instance}
    \begin{tabular}{@{}lrr@{}}
      \toprule
      Scenario & Total cost $Z$ & Acute-zone load \\
      \midrule
      Without chatbot & 7150 & 900 \\
      With chatbot & 2700 & 500 \\
      \textbf{Reduction} & \textbf{62\%} & \textbf{44\%} \\
      \bottomrule
    \end{tabular}
  \end{block}
  \vspace{-0.05cm}
  {\tiny Results from synthetic cohort routing \citep{thisstudy2026}.}
\end{frame}
