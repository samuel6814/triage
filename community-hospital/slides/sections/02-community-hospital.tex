\section{Community Hospital case study}
% =============================================================================

\begin{frame}{Case study setting}
  \small
  \textbf{Community Hospital} is a fictional Ghanaian LMIC facility (Kumasi-region style) \citep{Rominski2014, Gyedu2016}.

  \vspace{0.15cm}
  \begin{itemize}
    \item Synthetic cohort: \textbf{N = 1000} patients with \textbf{skewed ED arrival mix} (5\% / 15\% / 30\% / 25\% / 25\%) \citep{Mitchell2023, thisstudy2026}.
    \item Ghanaian patient-voice chief complaints (Kejetia, Bantama, Ejisu, Sunyani, KATH).
    \item Ground-truth nurse labels $c \in \{1,\ldots,5\}$ $\to$ $C = f_{\mathrm{SATS}}(c)$ $\to$ pathway $P(C)$ \citep{SATS2012, Rominski2014}.
    \item $\sim$7\% borderline complaints flagged for adjacent-level ambiguity \citep{thisstudy2026}.
    \item Chatbot eval: \textbf{97.5\%} colour-routing accuracy (975/1000); 8k holdout reference 99.92\% \citep{thisstudy2026, Lee2020}.
  \end{itemize}
\end{frame}

\begin{frame}{Green-pathway diversion}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \includegraphics[width=\textwidth]{green_diversion.png}
    \column{0.48\textwidth}
    \small
    \textbf{50\%} of cohort (levels 4--5, $n=500$) routed to OPD / Minors / Polyclinic \citep{thisstudy2026}.

    \vspace{0.2cm}
    Without early diversion, many Green patients queue in acute zones and worsen overcrowding \citep{Yilmaz2021, Mitchell2023}.

    \vspace{0.2cm}
    Chatbot assigns Green at intake, before the physical desk \citep{Sundstrom2014, thisstudy2026}.
  \end{columns}

\begin{frame}{Case study conclusion}
  \small
  \begin{block}{Findings}
    \begin{itemize}
      \item NLP maps Ghanaian free text to SATS colour before vitals are available \citep{Stewart2023, Gligorijevic2018, thisstudy2026}.
      \item Every case includes pathway destination and $T_{\max}$ \citep{Sundstrom2014, thisstudy2026}.
      \item 500/1000 patients are Green diversion candidates: operational lever on crowding \citep{Yilmaz2021, thisstudy2026}.
    \end{itemize}
  \end{block}

  \plain{Next: transshipment LP model for ED patient flow \citep{thisstudy2026}.}
\end{frame}
