\section{Methodology}

\begin{frame}{Methodology: Predict urgency from text}
	\footnotesize
	\setlength{\abovedisplayskip}{4pt}
	\setlength{\belowdisplayskip}{4pt}
	\begin{itemize}\setlength\itemsep{0.25em}
		\item \textbf{Tokenisation.} Split the English chief complaint into tokens (small pieces of the sentence), and place a special \texttt{[CLS]} token first.
		\item Each token becomes a vector by adding meaning, position, and a segment marker \citep{lee2020biobert}:
		\begin{equation}
		  E(t_i) = E_{\mathrm{word}}(t_i) + E_{\mathrm{pos}}(i) + E_{\mathrm{seg}}(0).
		  \tag{1}
		\end{equation}
		\item Attention is a weighted sum. Each token builds \(Q\) (what it looks for), \(K\) (the label it shows), and \(V\) (the meaning it passes on) \citep{vaswani2017attention}:
		\begin{equation}
		  \mathrm{Attention}(Q,K,V)
		  = \mathrm{softmax}\!\left(\frac{QK^{\top}}{\sqrt{d_k}}\right)V,
		  \quad d_k=64.
		  \tag{2}
		\end{equation}
		\(d_k\) is the length of each Key vector.
		The \texttt{[CLS]} vector at the end is the sentence summary used next.
	\end{itemize}
\end{frame}

\begin{frame}{Methodology: Colour and pathway at intake}
	\footnotesize
	\setlength{\abovedisplayskip}{4pt}
	\setlength{\belowdisplayskip}{4pt}
	\begin{itemize}\setlength\itemsep{0.25em}
		\item The \texttt{[CLS]} summary is scored into five logits \(\mathbf{z}\), the raw scores before softmax, one per urgency level:
		\begin{equation}
		  \mathbf{z} = W h_{\mathrm{[CLS]}} + \mathbf{b}.
		  \tag{3}
		\end{equation}
		\(\mathbf{z}\) is not a probability. A larger \(z_c\) means that urgency level looks stronger.
		\item Softmax turns those scores into five probabilities that add to 1. The largest probability is the urgency level, because the patient gets one colour:
		\begin{equation}
		  \hat{y}_c = \frac{\exp(z_c)}{\sum_{j=1}^{5}\exp(z_j)}.
		  \tag{4}
		\end{equation}
		\item A fixed map names the colour: \(1\to\)Red, \(2\to\)Orange, \(3\to\)Yellow, \(4\) or \(5\to\)Green \citep{satsmanual}.
	\end{itemize}
\end{frame}

\begin{frame}{Methodology: No colour for non-clinical chat}
	\footnotesize
	\setlength{\abovedisplayskip}{6pt}
	\setlength{\belowdisplayskip}{6pt}
	Before BioBERT runs, a gate scores how clinical the message is.
	\begin{itemize}\setlength\itemsep{0.4em}
		\item Symptom words raise the score. Non-clinical chat lowers it:
		\begin{equation}
		  g(X) = \mathrm{clip}\!\Bigl(\sum_j s_j(X) - p(X),\, 0,\, 1\Bigr).
		  \tag{5}
		\end{equation}
		\(s_j(X)\) is the score for symptom word \(j\). \(p(X)\) is the penalty if the message is non-clinical chat. \(\mathrm{clip}\) keeps the result between 0 and 1: below 0 becomes 0, above 1 becomes 1.
		\item A colour is given only if \(g(X)\geq 0.35\). Otherwise the system asks for a real complaint and stores no colour. This keeps non-clinical chat out of the classifier \citep{stewart2023nlp}.
	\end{itemize}
\end{frame}
