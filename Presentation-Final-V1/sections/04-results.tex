\section{Results}

\begin{frame}{Model Performance}
	\begin{block}{Holdout evaluation}
		Stratified 10\% holdout, \(N=8{,}000\) English Triagegeist complaints \citep{triagegeist2025}.
	\end{block}
	\begin{center}
	\small
	\begin{tabular}{|l|r|}
	\hline
	\textbf{Metric (baseline BioBERT)} & \textbf{Value} \\
	\hline
	Accuracy & 99.92\% \\
	Macro-F1 & 0.9977 \\
	Recall Level 1 (Red) & 98.14\% \\
	Mean latency & \(\approx\)102\,ms / sample \\
	\hline
	\end{tabular}
	\end{center}
	\medskip
	\begin{itemize}
		\item Strong agreement with nurse acuity labels on held-out English complaints.
		\item Fast enough for prototype intake assistance.
	\end{itemize}
\end{frame}
