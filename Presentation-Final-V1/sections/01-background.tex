\begin{frame}{Hospital Intake Under Pressure}
	\begin{itemize}
		\item Crowded waiting areas, limited nursing staff, long queues before first assessment.
		\item Teaching hospitals in Ghana, including \examplehospital{}, use the South African Triage Scale (SATS).
		\item Colour codes guide \emph{how fast} a patient must be seen and \emph{where} they stream next.
		\item \textbf{Acuity discontinuity:} urgency colour is often applied only at the first desk; after that it may no longer steer beds, labs, or re-assessment timers \citep{Rominski2014,kathsats}.
	\end{itemize}
\end{frame}

\begin{frame}{Text First, Then Vitals}
	\begin{block}{Chief complaint}
		Delays often start with \textbf{only words}: headache, fever, weakness, described in free text before any thermometer or pulse reading.
	\end{block}
	\begin{block}{TEWS (nurse standard)}
		HR, RR, mobility, temperature, AVPU, trauma \(\to\) score \(T\) and a SATS colour band \citep{satsmanual}.
	\end{block}
	\begin{exampleblock}{This project}
		English chief complaint \(\to\) SATS colour + pathway card before vitals; TEWS stays in parallel.
	\end{exampleblock}
\end{frame}

\begin{frame}{Literature Review}
	\small
	\begin{itemize}
		\item \textbf{Overcrowding and SATS.} LMIC intake pressure and acuity discontinuity; SATS colours with \(T_{\max}\) structure flow when urgency travels with the patient \citep{kathsats,satsmanual}. Text-first arrival remains under-formalised.
		\item \textbf{TEWS.} Vital-sign score \(T\) maps to a colour band; objective, and does not read free-text complaints \citep{satsmanual}.
		\item \textbf{Attention \(\to\) BioBERT.} Self-attention \(\to\) BERT; BioBERT continues on PubMed/PMC so clinical terms cluster in embedding space \citep{vaswani2017attention,lee2020biobert}.
		\item \textbf{NLP at triage.} Complaint classification is active research, mostly on high-income corpora; local SATS colour plus pathway mathematics is under-specified \citep{stewart2023nlp,gligorijevic2018deep}.
	\end{itemize}
\end{frame}
